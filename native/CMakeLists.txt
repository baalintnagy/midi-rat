cmake_minimum_required(VERSION 3.25)
project(midirat_bridge LANGUAGES CXX)
if(NOT WIN32)
  message(FATAL_ERROR "The MIDI Services bridge requires Windows")
endif()
add_library(midirat_bridge SHARED midirat_bridge.cpp)
target_compile_features(midirat_bridge PRIVATE cxx_std_20)
target_compile_definitions(midirat_bridge PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX)
target_compile_options(midirat_bridge PRIVATE /W4 /EHsc /permissive- /utf-8)
target_include_directories(midirat_bridge PRIVATE "${GENERATED_HEADERS}")
target_link_libraries(midirat_bridge PRIVATE windowsapp)
set_property(TARGET midirat_bridge PROPERTY MSVC_RUNTIME_LIBRARY MultiThreaded)
